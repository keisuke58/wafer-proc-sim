% ── 4. GP SURROGATE ──────────────────────────────────────────────────────────
\begin{frame}[shrink=5]{Process --- GP Surrogate for Chipping Prediction}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Heteroscedastic Gaussian Process \cite{williams2006gp}}
      \begin{itemize}
        \item Input: $(v_f,\, a_p,\, n_s)$; output: lateral chipping [$\mu$m]
        \item Noise variance fitted per observation\\
              (process noise $\neq$ const at high feeds)
        \item Kernel: Mat\'{e}rn-5/2 + white noise
        \item LOO cross-validation:
              \textbf{RMSE = 1.62~$\mu$m} on $n = 11$ Grade-A points
      \end{itemize}

      \vspace{0.5em}
      \begin{block}{Why GP?}
        \small
        Uncertainty quantification $\Rightarrow$ safe active-learning
        next-experiment selection (EI acquisition).
        New recipe candidates ranked \emph{before} cutting.
      \end{block}

      \vspace{0.5em}
      \begin{tabular}{ll}
        \toprule
        Metric & Value \\
        \midrule
        LOO RMSE & \textbf{1.62~$\mu$m} \\
        Training points & 11 \\
        Kernel & Mat\'{e}rn-5/2 \\
        \bottomrule
      \end{tabular}
    \end{column}

% ── NEW FRAME D: ACTIVE LEARNING ─────────────────────────────────────────────
\begin{frame}[shrink=5]{Process --- Active Learning for Next-Experiment Selection}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Expected Improvement (EI) Acquisition}
      \begin{itemize}
        \item EI acquisition guides next cut:
              \[
                \mathrm{EI}(x) = \mathbb{E}[\max(f^{*} - f(x),\,0)]
              \]
        \item GP posterior uncertainty drives exploration
              vs.\ exploitation trade-off
        \item Only \textbf{3 EI-selected experiments} needed to find
              recipe with chipping $< 5~\mu$m
        \item Compared to 11-point random grid:
              \textbf{64\% fewer experiments} required
      \end{itemize}

      \vspace{0.5em}
      \begin{block}{Practical Benefit}
        \small
        Each physical cut costs machine time and material.
        EI-guided selection reduces experimental campaign from
        $\sim$11 cuts to $\sim$3 --- critical for SiC's high
        wafer cost ($\sim$\$1,000 per 6-inch substrate).
      \end{block}
    \end{column}

% ── 5. PHYSICS-AWARE TRANSFORMER ─────────────────────────────────────────────
\begin{frame}[shrink=5]{R\&D --- Physics-Aware Transformer (PAT)}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Transfer Learning on Scarce Data}
      \begin{itemize}
        \item \textbf{Pre-train} on 2{,}000 FEM simulation points\\
              (stress field $\to$ chipping proxy label)
        \item \textbf{Fine-tune} discriminatively on $n=11$ real cuts
        \item LOO RMSE: \textbf{2.00~$\mu$m}

      \end{itemize}

      \vspace{0.4em}
      \textbf{Physics Features (hand-crafted)}
      \begin{itemize}
        \item \kv{MRR}{$= v_f \cdot a_p \cdot b$, material removal rate}
        \item \kv{SFR}{specific force ratio (tangential/normal)}
        \item \kv{4-fold anisotropy}{$\cos(4\theta)$ crystal orientation term}
      \end{itemize}

      \vspace{0.4em}
      \begin{alertblock}{Key Finding}
        \small
        The 4-fold SiC anisotropy feature reduces residual variance by
        $\approx 18\%$ vs.\ isotropic baseline.
      \end{alertblock}

% ── NEW FRAME E: BAYESIAN OPTIMISATION vs TMCMC ───────────────────────────────

% ── 6. CONDITIONAL DDPM ──────────────────────────────────────────────────────
\begin{frame}[shrink=5]{R\&D --- Conditional Diffusion Model (DDPM) for Recipe Generation}
  \begin{columns}[T]
    \begin{column}{0.53\textwidth}
      \textbf{Generative Model: $p(\text{recipe} \mid \text{target chipping})$
        \cite{ho2020ddpm}}
      \begin{itemize}
        \item Architecture: U-Net denoiser, 1{,}000 diffusion steps
        \item Conditioning: classifier-free guidance on target chipping
        \item Training: 1{,}000 epochs, cosine noise schedule

      \textbf{Operator Learning: sensors $\to$ $\sigma(x,y)$
        \cite{lu2021deeponet,raissi2019pinn}}
      \begin{itemize}
        \item \textbf{Branch net}: encodes spindle load / vibration sensors
        \item \textbf{Trunk net}: spatial coordinates $(x, y)$ on wafer
        \item Output: von-Mises stress field in real time
      \end{itemize}

      \vspace{0.4em}
      \textbf{Physics Loss (equilibrium residual)}
      \[
        \mathcal{L}_{\rm phys}
        = \bigl\|\nabla^2 \hat{\sigma}\bigr\|^2
        + \bigl\|\nabla\cdot\hat{\boldsymbol{\sigma}}\bigr\|^2
      \]

      \vspace{0.3em}
      \textbf{Results}
      \begin{itemize}
        \item RMSE $= 0.229~\text{MPa}$ (proof-of-concept)
        \item $R^2 = 0.386$ on FEM ground truth
        \item Inference: $< 1~\text{ms}$ per field
              vs.\ FEM: $\sim 30~\text{s}$
      \end{itemize}
